from flask import Flask, request
import numpy as np
import joblib

app = Flask(__name__)

model = joblib.load("model.pkl")

@app.route("/predict", methods=["POST"])
def predict():

    data = np.array(request.json["data"]).reshape(1,-1)

    pred = model.predict(data)

    return {"prediction": int(pred[0])}

app.run(debug=True)
